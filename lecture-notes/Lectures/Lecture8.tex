\section[Quantum Dots]{\hyperlink{toc}{Quantum Dots}}

\subsection{Introduction and Basic Concepts}

\begin{itemize}
    \item Quantum dots differ from single-electron transistors (SETs) due to their quantum confinement energy (\ylink{0:30}).
    \item Key features: quantized level spacing from quantum confinement, discrete energy levels.
    \item Requirements: relatively low number of electrons (semiconductors), small island.
    \item Can be built on different scales:
          \begin{itemize}
              \item Single molecule quantum dots (nm scale)
              \item Self-assembled quantum dots
              \item Gate-defined GaAs quantum dots
              \item Carbon nanotubes (micron scale)
              \item Graphene quantum dots
          \end{itemize}
\end{itemize}

\subsection{Quantum Dots vs Classical SETs}
\begin{itemize}
    \item Few-electron quantum dots in nanowires \parencite{Bjork2004}
    \item Classical dots: regular spacing of conductance peaks (\ylink{4:27}).
    \item Quantum dots: irregular peak spacing due to additional orbital energy.
    \item Addition energy includes both charging energy and quantum (orbital) energy (\ylink{6:30}).
    \item Current-voltage characteristics (\ylink{10:48}):
          \begin{itemize}
              \item Classical SET: zero current until threshold voltage, then linear current flow (because of continuous density of states)
              \item Quantum dot: step-like IV curve reflecting discrete level structure (because of discrete density of states)
          \end{itemize}
\end{itemize}

    \caption{SEM image of a double quantum dot device.}

\begin{itemize}
    \item Electrostatics of double quantum dots \parencite{vanderWiel2002,Hanson2007} see \ref{fig:double-quantum-dot-schematic}.

          \begin{figure}
              \centering
              \includegraphics[width=0.7\linewidth]{../Images/double-dot-circuit-diagram-vanderwiel2002.png}
              \caption{Schematic of a double quantum dot system \parencite{vanderWiel2002}}
              \label{fig:double-quantum-dot-schematic}
          \end{figure}


    \item Honeycomb charge stability diagrams (\ylink{56:00}) see \ref{fig:double-quantum-dot-honeycomb}.
          \begin{figure}
              \centering
              \includegraphics[width=0.5\linewidth]{../Images/CSD-double-dot-coupling-vanderWiel2002.png}
              \caption{Honeycomb charge stability diagram of a double quantum dot \parencite{vanderWiel2002}.}
              \label{fig:double-quantum-dot-honeycomb}
          \end{figure}
          \begin{itemize}
              \item Now conduction only happens at Triple points: points where three charge states are degenerate \parencite{Pothier1992}
              \item You can use the Stoof and Nazarov model to extract tunnel rates and inter-dot coupling from these triangles \parencite{Stoof1996}
                    \begin{equation}
                        I(\epsilon) = \frac{e}{h} \frac{\Gamma_R |T_{LR}|^2}{\epsilon^2 + \Gamma_R^2/4 + |T_{LR}|^2 (2 + \Gamma_R/\Gamma_L)}
                    \end{equation}
          \end{itemize}

    \item DQDs enable study of:
          \begin{itemize}
              \item Distinguishing from single quantum dot, here we two latters of discrete quantum states and the tunnel coupling is only determined by the inner transition without the effects of the outer thermally broadened leads \parencite{Kouwenhoven1995,VanderVaart1995}
              \item Much more sensitive to very narrow transitions
              \item Co-tunneling processes
              \item Finite bias triangles: regions of allowed transport under finite bias (\ylink{1:05:19})
          \end{itemize}
    \item Can also make triple quantum dots but then the stability diagram becomes 3D \parencite{Schroeder2007}.
\end{itemize}
